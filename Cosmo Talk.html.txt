            <!-- 자바스크립트가 소설 내용을 카톡 양식으로 변환해서 채워줍니다 -->

    <script>
        // ==========================================
        // ✍️ 여기에 소설 대화 내용을 작성하세요!
        // 형식: **이름** | "내용" **시간**
        // ==========================================
        const novelChatScript = `
**🌸코스모톡**
**바인** | "아리아 벨로모니아, 좋은 아침이야. 어제 많이 마신 것 같더라. 문 앞에 속 편한 음식이랑 배 스무디 보내놨어." **A.M 07:45**
**바인** | "푹 쉬고, 이따 저녁 7시에 시계탑에서 보자." **A.M 07:47**
        `;
        // ==========================================

        function renderChat() {
            const chatBody = document.getElementById('chatBody');
            const lines = novelChatScript.trim().split(/\r?\n/);
            
            lines.forEach(line => {
                const trimmed = line.trim();
                if (!trimmed || trimmed.startsWith('**🌸코스모톡**')) return;

                // 파싱 규칙: **이름** | "대화내용" **시간**
                const parts = trimmed.split('|');
                if (parts.length < 2) return;

                let sender = parts[0].replace(/\*\*/g, '').trim();
                let rest = parts[1].trim();

                // 시간 추출 (마지막 **시간** 패턴 찾기)
                let time = '';
                const timeMatch = rest.match(/\*\*(.*?)\*\*/g);
                if (timeMatch && timeMatch.length > 0) {
                    const lastTimeTag = timeMatch[timeMatch.length - 1];
                    time = lastTimeTag.replace(/\*\*/g, '');
                    rest = rest.replace(lastTimeTag, '').trim();
                }

                // 대화 내용 양쪽 따옴표 제거 및 띄어쓰기 유지 처리
                let message = rest.replace(/^["']|["']$/g, '').trim();

                // 내가 보낸 대화인지 여부 설정 (원하는 캐릭터 이름을 '나'로 지정 가능)
                const isMe = (sender === '나'); // 본인 이름이 '나'가 아니라면 여기서 비교군 수정 가능

                // HTML 요소 생성
                const messageRow = document.createElement('div');
                messageRow.className = `message-row ${isMe ? 'me' : ''}`;

                messageRow.innerHTML = `
                    <div class="profile-icon">🌸</div>
                    <div class="message-content">
                        <div class="sender-name">${sender}</div>
                        <div class="bubble-wrapper">
                            <div class="bubble">${message}</div>
                            <div class="time">${time}</div>
                        </div>
                    </div>
                `;

                chatBody.appendChild(messageRow);
            });
        }

        window.onload = renderChat;
    </script>
